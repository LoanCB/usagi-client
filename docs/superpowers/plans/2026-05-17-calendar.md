# Calendar View — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter une vue Calendrier (toggle Mois/Semaine) dans la sidebar, affichant les tâches par `dueDate` et `completedAt`, avec création inline par clic sur un jour et navigation vers le détail par clic sur une tâche.

**Architecture:** `CalendarView` appelle `loadTasks(repo, { allTasks: true })` au mount et lit depuis `useTaskStore`, comme `TaskList`. Une fonction pure `groupTasksByDate` groupe les tâches par date. Trois sous-composants couvrent chaque responsabilité : `CalendarHeader`, `MonthView`, `WeekView`.

**Tech Stack:** React 19, TypeScript, date-fns v4, Tailwind v4, lucide-react, Zustand, Vitest, Playwright

      dueDate: "2026-05-10",

    expect(map.get("2026-05-10")).toBeUndefined();

import type { Task } from "@/types";

export function groupTasksByDate(
  tasks: Task[],
): Map<string, { due: Task[]; completed: Task[] }> {
  const map = new Map<string, { due: Task[]; completed: Task[] }>();

  function getOrCreate(date: string) {
    if (!map.has(date)) map.set(date, { due: [], completed: [] });
    const entry = map.get(date);
    if (!entry) throw new Error("unreachable");
    return entry;
  }

  for (const task of tasks) {
    if (task.completedAt) {
      getOrCreate(task.completedAt.slice(0, 10)).completed.push(task);
    } else if (task.dueDate) {
      getOrCreate(task.dueDate).due.push(task);
    }
  }

  return map;
}
```

import { useRef, useState } from "react";
import { useTranslation } from "react-i18next";
import { TagSelector } from "@/components/tasks/TagSelector";
import { getRepository } from "@/store/repository";
import { useTaskStore } from "@/store/tasks";

interface QuickAddTaskProps {
  readonly projectId: string | null | undefined;
  readonly dueDate?: string | null;
}

export function QuickAddTask({ projectId, dueDate }: QuickAddTaskProps) {
  const [title, setTitle] = useState("");
  const [tagIds, setTagIds] = useState<string[]>([]);
  const inputRef = useRef<HTMLInputElement>(null);
  const createTask = useTaskStore((s) => s.createTask);
  const { t } = useTranslation();

  async function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === "Enter") {
      const trimmed = title.trim();
      if (!trimmed) return;
      try {
        await createTask(getRepository(), {
          title: trimmed,
          projectId: projectId ?? null,
          tagIds,
          ...(dueDate !== undefined && dueDate !== null ? { dueDate } : {}),
        });
        setTitle("");
        setTagIds([]);
        inputRef.current?.focus();
      } catch (err) {
        console.error("Failed to create task", err);
      }
    } else if (e.key === "Escape") {
      setTitle("");
    }
  }

  return (
    <div className="flex items-center gap-3 mx-3 mb-3 mt-1 px-4 py-2.5 rounded-xl glass-stat">
      <div className="w-3.5 h-3.5 rounded-sm border border-dashed border-muted-foreground/40 shrink-0" />
      <input
        ref={inputRef}
        type="text"
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        onKeyDown={handleKeyDown}
        placeholder={t("task.titlePlaceholder")}
        aria-label={t("task.titlePlaceholder")}
        className="flex-1 bg-transparent text-sm text-foreground placeholder:text-muted-foreground/50 outline-none"
      />
      <TagSelector
        selectedTagIds={tagIds}
        onChange={setTagIds}
        triggerClassName="text-muted-foreground"
        projectId={projectId}
      />
    </div>
  );
}
```

interface MonthViewProps {

  readonly grouped: Map<string, { due: Task[]; completed: Task[] }>;

  readonly onDayClick: (date: string) => void;

export function MonthView({
  currentDate,
  grouped,
  selectedDay,
  onDayClick,
  onTaskClick,
}: MonthViewProps) {
  const { i18n } = useTranslation();
  const locale = i18n.language === "fr" ? fr : enUS;

  const monthStart = startOfMonth(currentDate);
  const calStart = startOfWeek(monthStart, { weekStartsOn: 1 });
  const calEnd = endOfWeek(endOfMonth(currentDate), { weekStartsOn: 1 });
  const days = eachDayOfInterval({ start: calStart, end: calEnd });

  // Day header labels: L M M J V S D
  const dayHeaders = Array.from({ length: 7 }, (_, i) => {

    d.setDate(d.getDate() + i);

  return (
    <div className="flex flex-col flex-1 overflow-auto px-4 pb-4">
      {/* Day headers */}

          <div
            key={label}
            className="text-center text-xs font-semibold text-muted-foreground/60 uppercase tracking-wider py-1"
          >
            {label}
          </div>
        ))}
      </div>

      {/* Day grid */}
      <div className="grid grid-cols-7 gap-px flex-1">
        {days.map((day) => {
          const iso = format(day, "yyyy-MM-dd");
          const entry = grouped.get(iso);
          const due = entry?.due ?? [];
          const completed = entry?.completed ?? [];
          const today = isToday(day);
          const sameMonth = isSameMonth(day, currentDate);
          const selected = selectedDay === iso;

          return (
            <div
              key={iso}
              onClick={() => onDayClick(iso)}
              className={cn(
                "min-h-[90px] border border-border/20 p-1.5 cursor-pointer transition-colors hover:bg-foreground/5",
                !sameMonth && "opacity-40",
                today && "bg-primary/10 border-primary/30",

              <div
                className={cn(
                  "text-xs font-medium mb-1.5 w-6 h-6 flex items-center justify-center rounded-full",

                  !today && "text-foreground/70",
                )}
              >
                {format(day, "d")}
              </div>
              <div className="flex flex-wrap gap-1">
                {due.map((task) => (
                  <button
                    key={task.id}
                    type="button"
                    title={task.title}
                    onClick={(e) => {
                      e.stopPropagation();
                      onTaskClick(task);
                    }}
                    className="w-2.5 h-2.5 rounded-full bg-orange-400/80 hover:scale-125 transition-transform shrink-0"

                      onTaskClick(task);
                    }}
                    className="w-2.5 h-2.5 rounded-full bg-green-400/80 hover:scale-125 transition-transform shrink-0"
                  />
                ))}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

                  onClick={(e) => {

                >

            </div>

  );

- [ ] **Step 1: Créer `src/components/calendar/CalendarView.tsx`**

```tsx
import { addMonths, addWeeks, subMonths, subWeeks } from "date-fns";
import { useMemo, useState, useEffect } from "react";
import { useTranslation } from "react-i18next";
import { QuickAddTask } from "@/components/tasks/QuickAddTask";
import { groupTasksByDate } from "@/lib/calendarUtils";
import { getRepository } from "@/store/repository";
import { useTaskStore } from "@/store/tasks";
import { useUIStore } from "@/store/ui";
import type { Task } from "@/types";
import { CalendarHeader, type CalendarViewMode } from "./CalendarHeader";
import { MonthView } from "./MonthView";
import { WeekView } from "./WeekView";

export function CalendarView() {
  const [viewMode, setViewMode] = useState<CalendarViewMode>("month");
  const [currentDate, setCurrentDate] = useState(() => new Date());
  const [selectedDay, setSelectedDay] = useState<string | null>(null);

  const { loadTasks } = useTaskStore();
  const tasks = useTaskStore((s) => s.tasks);
  const { setSelectedProject, setSelectedTask } = useUIStore();

  useEffect(() => {
    loadTasks(getRepository(), { allTasks: true });
  }, [loadTasks]);

  const grouped = useMemo(() => groupTasksByDate(tasks), [tasks]);

  function handlePrev() {
    setCurrentDate((d) =>
      viewMode === "month" ? subMonths(d, 1) : subWeeks(d, 1),
    );
    setSelectedDay(null);
  }

  function handleNext() {
    setCurrentDate((d) =>
      viewMode === "month" ? addMonths(d, 1) : addWeeks(d, 1),
    );
    setSelectedDay(null);
  }

  function handleDayClick(date: string) {
    setSelectedDay((prev) => (prev === date ? null : date));
  }

  function handleTaskClick(task: Task) {
    setSelectedProject(task.projectId ?? undefined);
    setSelectedTask(task.id);
  }

  return (
    <div className="flex flex-col flex-1 min-w-0 overflow-hidden">
      <CalendarHeader
        currentDate={currentDate}
        viewMode={viewMode}
        onViewModeChange={(mode) => {
          setViewMode(mode);
          setSelectedDay(null);
        }}
        onPrev={handlePrev}
        onNext={handleNext}
      />

      {viewMode === "month" ? (
        <MonthView
          currentDate={currentDate}
          grouped={grouped}
          selectedDay={selectedDay}
          onDayClick={handleDayClick}
          onTaskClick={handleTaskClick}
        />
      ) : (
        <WeekView
          currentDate={currentDate}
          grouped={grouped}
          selectedDay={selectedDay}
          onDayClick={handleDayClick}
          onTaskClick={handleTaskClick}
        />
      )}

      {selectedDay && (
        <div className="shrink-0 border-t border-border/40 px-4 py-2">
          <QuickAddTask projectId={null} dueDate={selectedDay} />
        </div>
      )}
    </div>
  );
}
```
